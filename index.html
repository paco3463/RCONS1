    <!-- Three.js Setup and Animation -->

    <script>

        document.addEventListener('DOMContentLoaded', function() {

            initThreeJS();

            initNavbarScroll();

        });


        // Header Background Change on Scroll

        function initNavbarScroll() {

            const header = document.querySelector('header');

            const logoText = document.querySelectorAll('.logo-text-main, .logo-text-sub');

            const navLinks = document.querySelectorAll('nav a');

            const mobileBtn = document.querySelector('button.md\\:hidden');


            window.addEventListener('scroll', () => {

                if (window.scrollY > 50) {

                    header.classList.add('bg-white', 'shadow-md', 'py-4');

                    header.classList.remove('py-8');

                    // Change text to dark

                    logoText.forEach(el => el.classList.replace('text-white', 'text-brandDark'));

                    navLinks.forEach(el => el.classList.replace('text-white', 'text-brandDark'));

                    mobileBtn.classList.replace('text-white', 'text-brandDark');

                } else {

                    header.classList.remove('bg-white', 'shadow-md', 'py-4');

                    header.classList.add('py-8');

                    // Change text back to white

                    logoText.forEach(el => el.classList.replace('text-brandDark', 'text-white'));

                    navLinks.forEach(el => el.classList.replace('text-brandDark', 'text-white'));

                    mobileBtn.classList.replace('text-brandDark', 'text-white');

                }

            });

        }


        // Three.js Wireframe Home Implementation

        function initThreeJS() {

            const container = document.getElementById('canvas-container');

            

            // Scene Setup

            const scene = new THREE.Scene();

            

            // Camera Setup

            const camera = new THREE.PerspectiveCamera(45, container.clientWidth / container.clientHeight, 0.1, 1000);

            camera.position.set(10, 8, 12);

            camera.lookAt(0, 0, 0);


            // Renderer Setup

            const renderer = new THREE.WebGLRenderer({ antialias: true, alpha: true });

            renderer.setSize(container.clientWidth, container.clientHeight);

            renderer.setPixelRatio(window.devicePixelRatio);

            container.appendChild(renderer.domElement);


            // Group to hold our house architecture

            const houseGroup = new THREE.Group();


            // Material - Using the requested #ff7745 accent color

            const wireMaterial = new THREE.LineBasicMaterial({ 

                color: 0xff7745,

                linewidth: 1,

                transparent: true,

                opacity: 1.0 // Increased opacity to 1.0 for better visibility

            });


            // Helper function to create wireframe boxes

            function createWireBox(width, height, depth, x, y, z) {

                const geometry = new THREE.BoxGeometry(width, height, depth);

                const edges = new THREE.EdgesGeometry(geometry);

                const line = new THREE.LineSegments(edges, wireMaterial);

                line.position.set(x, y, z);

                return line;

            }


            // Build a modern house structure (abstract intersecting volumes)

            

            // Ground floor main volume

            houseGroup.add(createWireBox(6, 2, 4, 0, 1, 0));

            // Ground floor secondary volume (garage/wing)

            houseGroup.add(createWireBox(3, 1.8, 3, -4.5, 0.9, 0.5));

            

            // Second floor main volume (cantilevered)

            houseGroup.add(createWireBox(5, 1.5, 3.5, 0.5, 2.75, 0.5));

            

            // Vertical architectural feature (chimney/stairwell)

            houseGroup.add(createWireBox(1, 4.5, 1.5, 2.5, 2.25, -1));


            // Pool / Terrace outline

            const poolGeo = new THREE.PlaneGeometry(8, 4);

            const poolEdges = new THREE.EdgesGeometry(poolGeo);

            const poolLine = new THREE.LineSegments(poolEdges, wireMaterial);

            poolLine.rotation.x = -Math.PI / 2;

            poolLine.position.set(1, 0, 4);

            houseGroup.add(poolLine);


            // Add subtle grid helper on the ground

            const gridHelper = new THREE.GridHelper(20, 20, 0xdddddd, 0xeeeeee);

            gridHelper.position.y = 0;

            scene.add(gridHelper);


            scene.add(houseGroup);


            // Mouse interaction logic

            let isDragging = false;

            let previousMousePosition = { x: 0, y: 0 };

            

            container.addEventListener('mousedown', (e) => {

                isDragging = true;

            });

            

            container.addEventListener('mousemove', (e) => {

                if (isDragging) {

                    const deltaMove = {

                        x: e.offsetX - previousMousePosition.x,

                        y: e.offsetY - previousMousePosition.y

                    };

                    

                    houseGroup.rotation.y += deltaMove.x * 0.005;

                    houseGroup.rotation.x += deltaMove.y * 0.005;

                    

                    // Limit X rotation so it doesn't flip upside down

                    houseGroup.rotation.x = Math.max(-Math.PI/4, Math.min(Math.PI/4, houseGroup.rotation.x));

                }

                

                previousMousePosition = {

                    x: e.offsetX,

                    y: e.offsetY

                };

            });

            

            window.addEventListener('mouseup', () => { isDragging = false; });

            

            // Touch support for mobile

            container.addEventListener('touchstart', (e) => {

                isDragging = true;

                previousMousePosition = { x: e.touches[0].clientX, y: e.touches[0].clientY };

            }, {passive: true});

            

            container.addEventListener('touchmove', (e) => {

                if (isDragging) {

                    const deltaMove = {

                        x: e.touches[0].clientX - previousMousePosition.x,

                        y: e.touches[0].clientY - previousMousePosition.y

                    };

                    houseGroup.rotation.y += deltaMove.x * 0.005;

                    houseGroup.rotation.x += deltaMove.y * 0.005;

                    houseGroup.rotation.x = Math.max(-Math.PI/4, Math.min(Math.PI/4, houseGroup.rotation.x));

                }

                previousMousePosition = { x: e.touches[0].clientX, y: e.touches[0].clientY };

            }, {passive: true});

            

            window.addEventListener('touchend', () => { isDragging = false; });


            // Animation Loop

            const animate = function () {

                requestAnimationFrame(animate);

                

                // Slow continuous rotation if not interacting

                if (!isDragging) {

                    houseGroup.rotation.y += 0.002;

                }


                renderer.render(scene, camera);

            };


            animate();


            // Handle Resize

            window.addEventListener('resize', () => {

                const width = container.clientWidth;

                const height = container.clientHeight;

                renderer.setSize(width, height);

                camera.aspect = width / height;

                camera.updateProjectionMatrix();

            });

        }

    </script>
